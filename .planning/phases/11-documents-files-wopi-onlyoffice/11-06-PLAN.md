**useGlobalSearch.ts:**

Create a TanStack Query hook for the global search API:

```typescript
import { useQuery } from '@tanstack/react-query'
import type { GlobalSearchResponse } from '@/api/types/document-types'

interface UseGlobalSearchOptions {
  query: string
  modules?: string[]  // filter to specific modules
  limit?: number      // results per module (default 5)
  enabled?: boolean
}

export function useGlobalSearch({ query, modules, limit = 5, enabled = true }: UseGlobalSearchOptions) {
  return useQuery<GlobalSearchResponse>({
    queryKey: ['global-search', query, modules, limit],
    queryFn: async () => {
      const params = new URLSearchParams({ q: query, limit: String(limit) })
      if (modules?.length) params.set('modules', modules.join(','))
      const response = await fetch(`${BASE_URL}/api/v1/search?${params}`, {
        headers: { Authorization: `Bearer ${getToken()}` }
      })
      if (!response.ok) throw new Error('Search failed')
      return response.json()
    },
    enabled: enabled && query.length >= 2,  // minimum 2 chars
    staleTime: 30_000,  // 30s cache
    gcTime: 60_000,     // 1 min garbage collection
  })
}
```

])

function isOnlyOfficeEditable(mimeType: string): boolean {
