**Migration 000031_create_time_entries.up.sql:**

```sql
-- Time entries table for task time tracking
-- Supports both timer-based (start/stop) and manual entries

CREATE TABLE time_entries (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    task_id UUID NOT NULL REFERENCES tasks(id) ON DELETE CASCADE,
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    started_at TIMESTAMPTZ NOT NULL,
    ended_at TIMESTAMPTZ,                       -- NULL = timer still running
    duration_seconds INTEGER,                   -- computed on stop; NULL while running; set on manual entry
    description TEXT,                            -- optional note for this time entry
    is_manual BOOLEAN NOT NULL DEFAULT false,    -- true for manually entered time, false for timer
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

-- Core indexes
CREATE INDEX idx_time_entries_task ON time_entries (task_id);
CREATE INDEX idx_time_entries_user ON time_entries (user_id);
CREATE INDEX idx_time_entries_started ON time_entries (started_at DESC);
-- Index for finding the active timer (where ended_at IS NULL)
CREATE INDEX idx_time_entries_active ON time_entries (user_id, ended_at) WHERE ended_at IS NULL;
```

```go
    ListByUser(ctx context.Context, userID uuid.UUID, page, pageSize int) ([]*models.TimeEntry, int, error)
    GetActiveTimer(ctx context.Context, userID uuid.UUID) (*models.ActiveTimer, error)

    GetTaskTimeSummary(ctx context.Context, taskID uuid.UUID) (*models.TimeEntrySummary, error)
}
```

Implementation with pgx pool:
- `Create`: INSERT INTO time_entries
- `GetByID`: SELECT with id
- `Update`: UPDATE SET duration_seconds, description, updated_at (for editing manual entries)
- `Delete`: DELETE by id (only if user_id matches)
- `ListByTask`: SELECT ... WHERE task_id ORDER BY started_at DESC with pagination
- `ListByUser`: SELECT ... WHERE user_id ORDER BY started_at DESC with pagination
- `GetActiveTimer`: SELECT te.*, t.title AS task_title, p.project_key, t.task_number FROM time_entries te JOIN tasks t ON te.task_id = t.id LEFT JOIN projects p ON t.project_id = p.id WHERE te.user_id = $1 AND te.ended_at IS NULL LIMIT 1
- `StopActiveTimer`: UPDATE time_entries SET ended_at = NOW(), duration_seconds = EXTRACT(EPOCH FROM (NOW() - started_at))::INTEGER, updated_at = NOW() WHERE user_id = $1 AND ended_at IS NULL RETURNING *
- `GetTaskTimeSummary`: SELECT task_id, SUM(duration_seconds) as total, COUNT(*) as count FROM time_entries WHERE task_id = $1 AND duration_seconds IS NOT NULL GROUP BY task_id
