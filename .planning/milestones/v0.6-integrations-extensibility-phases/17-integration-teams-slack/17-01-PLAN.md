    Create migration 000053 with 5 tables following the schema from 17-RESEARCH.md:

    **integration_configs** -- one per platform per org:
    - id UUID PK, platform VARCHAR(20) CHECK IN ('teams','slack'), is_active BOOLEAN DEFAULT false
    - credentials_vault_key VARCHAR(200) NOT NULL (references vault secret for encrypted storage)
    - metadata JSONB DEFAULT '{}' (team_id for Slack, tenant_id for Teams)
    - created_by UUID FK users(id), created_at/updated_at TIMESTAMPTZ
    - UNIQUE(platform)

    **integration_channel_mappings** -- module-to-channel routing:
    - id UUID PK, config_id UUID FK integration_configs ON DELETE CASCADE
    - channel_id VARCHAR(200) NOT NULL, channel_name VARCHAR(200) NOT NULL
    - modules JSONB DEFAULT '[]' (array of module IDs: "crm", "hr", "biz", "work", etc.)
    - is_active BOOLEAN DEFAULT true
    - platform_data JSONB DEFAULT '{}' (conversation reference for Teams, etc.)
    - created_at/updated_at TIMESTAMPTZ
    - INDEX idx_integration_mappings_config(config_id)
    - Partial INDEX idx_integration_mappings_active(config_id) WHERE is_active = true

    **integration_account_links** -- external user to KMU Hub user mapping:
    - id UUID PK, platform VARCHAR(20) CHECK IN ('teams','slack')
    - external_user_id VARCHAR(200) NOT NULL, kmuhub_user_id UUID FK users(id)
    - external_display_name VARCHAR(200)
    - linked_at TIMESTAMPTZ DEFAULT NOW()
    - UNIQUE(platform, external_user_id), UNIQUE(platform, kmuhub_user_id)
    - INDEX idx_account_links_lookup(platform, external_user_id)

    **integration_delivery_log** -- forwarded notification audit trail:
    - id UUID PK, notification_id UUID NOT NULL
    - mapping_id UUID FK integration_channel_mappings(id)
    - platform VARCHAR(20) NOT NULL, status VARCHAR(20) CHECK IN ('sent','failed','rate_limited')
    - platform_message_id VARCHAR(200), error_message TEXT
    - created_at TIMESTAMPTZ DEFAULT NOW()
    - INDEX idx_delivery_log_notification(notification_id)
    - Partial INDEX idx_delivery_log_cleanup(created_at) WHERE created_at < NOW() - INTERVAL '30 days'

    **integration_link_tokens** -- short-lived tokens for account linking flow:
    - id UUID PK, platform VARCHAR(20) NOT NULL
    - external_user_id VARCHAR(200) NOT NULL
    - token_hash VARCHAR(64) NOT NULL (SHA-256)
    - expires_at TIMESTAMPTZ NOT NULL, used BOOLEAN DEFAULT false
    - created_at TIMESTAMPTZ DEFAULT NOW()
    - Partial INDEX idx_link_tokens_hash(token_hash) WHERE NOT used

    Down migration: DROP all 5 tables in reverse order (link_tokens, delivery_log, account_links, channel_mappings, configs).

    **repository.go** -- Define Repository interface:
    ```go
    type Repository interface {
        // Config CRUD
        CreateConfig(ctx context.Context, cfg *IntegrationConfig) error
        GetConfigByPlatform(ctx context.Context, platform string) (*IntegrationConfig, error)
        UpdateConfig(ctx context.Context, cfg *IntegrationConfig) error
        ListConfigs(ctx context.Context) ([]*IntegrationConfig, error)
        DeleteConfig(ctx context.Context, id uuid.UUID) error

        // Channel Mapping CRUD
        CreateMapping(ctx context.Context, m *ChannelMapping) error
        GetMapping(ctx context.Context, id uuid.UUID) (*ChannelMapping, error)
        ListMappingsByConfig(ctx context.Context, configID uuid.UUID) ([]*ChannelMapping, error)
        ListActiveMappingsForModule(ctx context.Context, moduleID string) ([]*ChannelMapping, error)
        UpdateMapping(ctx context.Context, m *ChannelMapping) error
        DeleteMapping(ctx context.Context, id uuid.UUID) error

        // Account Links
        CreateAccountLink(ctx context.Context, link *AccountLink) error
        GetAccountLink(ctx context.Context, platform, externalUserID string) (*AccountLink, error)
        GetAccountLinkByKMUHubUser(ctx context.Context, platform string, kmuhubUserID uuid.UUID) (*AccountLink, error)
        DeleteAccountLink(ctx context.Context, id uuid.UUID) error

        // Link Tokens
        CreateLinkToken(ctx context.Context, token *LinkToken) error
        GetLinkTokenByHash(ctx context.Context, tokenHash string) (*LinkToken, error)
        MarkLinkTokenUsed(ctx context.Context, id uuid.UUID) error
        CleanupExpiredTokens(ctx context.Context) (int, error)

        // Delivery Log
        LogDelivery(ctx context.Context, entry *DeliveryLogEntry) error
        GetRecentFailures(ctx context.Context, mappingID uuid.UUID, limit int) ([]*DeliveryLogEntry, error)
        CleanupOldLogs(ctx context.Context, olderThan time.Time) (int, error)
    }
    ```

    **postgres_repository.go** -- PostgreSQL implementation:
    - Constructor: `NewPostgresRepository(pool *pgxpool.Pool) *PostgresRepository`
    - Implement all Repository interface methods using pgx with $1-style parameterized queries
    - ListActiveMappingsForModule: query `integration_channel_mappings` JOIN `integration_configs` WHERE is_active=true AND modules JSONB contains the module ID (`modules @> $1::jsonb`)
    - GetAccountLink: SELECT by platform + external_user_id
    - CreateAccountLink: INSERT with ON CONFLICT DO UPDATE on (platform, external_user_id) for upsert semantics (single active link per external user per platform, per research pitfall #4)
    - CleanupExpiredTokens: DELETE WHERE expires_at < NOW() AND NOT used
    - CleanupOldLogs: DELETE WHERE created_at < $1
