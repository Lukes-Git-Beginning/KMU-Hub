    **2. Create `desktop/src/renderer/src/api/hr-client.ts`** following the pattern from `calendar-client.ts` and `video-client.ts` (fetch wrapper, not openapi-fetch):

    ```typescript
    import { getAuthHeaders, handleApiError } from './client'

    const API_BASE = '/api/v1/hr'

    async function hrFetch<T>(path: string, options?: RequestInit): Promise<T> {
        const res = await fetch(`${API_BASE}${path}`, {
            ...options,
            headers: { ...getAuthHeaders(), 'Content-Type': 'application/json', ...options?.headers },
        })
        if (!res.ok) await handleApiError(res)
        return res.json()
    }

    // Export typed API methods for each endpoint group
    export const hrClient = {
        // Leave
        createLeaveRequest: (data: CreateLeaveRequestInput) => hrFetch<LeaveRequest>('/leave/requests', { method: 'POST', body: JSON.stringify(data) }),
        listLeaveRequests: (params?: Record<string, string>) => hrFetch<{ requests: LeaveRequest[]; total: number }>(`/leave/requests?${new URLSearchParams(params)}`),
        getLeaveRequest: (id: string) => hrFetch<LeaveRequest>(`/leave/requests/${id}`),
        approveLeaveRequest: (id: string, data?: ApproveRejectInput) => hrFetch<void>(`/leave/requests/${id}/approve`, { method: 'POST', body: JSON.stringify(data ?? {}) }),
        rejectLeaveRequest: (id: string, data?: ApproveRejectInput) => hrFetch<void>(`/leave/requests/${id}/reject`, { method: 'POST', body: JSON.stringify(data ?? {}) }),
        cancelLeaveRequest: (id: string) => hrFetch<void>(`/leave/requests/${id}/cancel`, { method: 'POST' }),
        getLeaveBalance: () => hrFetch<LeaveBalance>('/leave/balance'),
        getEmployeeLeaveBalance: (userId: string) => hrFetch<LeaveBalance>(`/leave/balance/${userId}`),
        listLeaveTypes: () => hrFetch<{ types: LeaveType[] }>('/leave/types'),
        recordSickLeave: (data: RecordSickLeaveInput) => hrFetch<LeaveRequest>('/leave/sick', { method: 'POST', body: JSON.stringify(data) }),
        // Time tracking
        clockIn: () => hrFetch<{ entry: WorkTimeEntry; compliance: ArbZGComplianceResult }>('/time/clock-in', { method: 'POST' }),
        clockOut: () => hrFetch<{ entry: WorkTimeEntry; compliance: ArbZGComplianceResult }>('/time/clock-out', { method: 'POST' }),
        startBreak: () => hrFetch<BreakEntry>('/time/break/start', { method: 'POST' }),
        endBreak: () => hrFetch<BreakEntry>('/time/break/end', { method: 'POST' }),
        getActiveShift: () => hrFetch<{ entry?: WorkTimeEntry; breaks: BreakEntry[] }>('/time/active'),
        getWorkTimeStatus: () => hrFetch<WorkTimeStatus>('/time/status'),
        listWorkTimeEntries: (params?: Record<string, string>) => hrFetch<{ entries: WorkTimeEntry[]; total: number }>(`/time/entries?${new URLSearchParams(params)}`),
        getDailySummary: (date: string) => hrFetch<DailySummary>(`/time/summary/daily?date=${date}`),
        getWeeklySummary: (weekStart: string) => hrFetch<WeeklySummary>(`/time/summary/weekly?week_start=${weekStart}`),
        submitCorrection: (data: SubmitCorrectionInput) => hrFetch<WorkTimeEntry>('/time/corrections', { method: 'POST', body: JSON.stringify(data) }),
        approveCorrection: (id: string) => hrFetch<WorkTimeEntry>(`/time/corrections/${id}/approve`, { method: 'POST' }),
        // Absences
        getAbsenceCalendar: (params: Record<string, string>) => hrFetch<{ entries: AbsenceEntry[] }>(`/absences/calendar?${new URLSearchParams(params)}`),
        // Employees
        listEmployees: (params?: Record<string, string>) => hrFetch<{ employees: EmployeeProfile[]; total: number }>(`/employees?${new URLSearchParams(params)}`),
        getSelfProfile: () => hrFetch<EmployeeProfile>('/employees/me'),
        updateSelfProfile: (data: UpdateSelfProfileInput) => hrFetch<EmployeeProfile>('/employees/me', { method: 'PUT', body: JSON.stringify(data) }),
        getEmployee: (id: string) => hrFetch<EmployeeProfile>(`/employees/${id}`),
        updateEmployee: (id: string, data: UpdateEmployeeInput) => hrFetch<EmployeeProfile>(`/employees/${id}`, { method: 'PUT', body: JSON.stringify(data) }),
        listEmployeeDocuments: (id: string) => hrFetch<{ documents: EmployeeDocument[] }>(`/employees/${id}/documents`),
        uploadEmployeeDocument: (id: string, data: UploadDocumentInput) => hrFetch<EmployeeDocument>(`/employees/${id}/documents`, { method: 'POST', body: JSON.stringify(data) }),
        listDocumentCategories: (id: string) => hrFetch<{ categories: HRDocumentCategory[] }>(`/employees/${id}/documents/categories`),
        // Settings
        getHRSettings: () => hrFetch<HRSettings>('/settings'),
        updateHRSettings: (data: Partial<HRSettings>) => hrFetch<HRSettings>('/settings', { method: 'PUT', body: JSON.stringify(data) }),
    }
    ```

    Query key pattern: `['hr', domain, ...params]` (consistent with `['finance', ...]` from Phase 12).

    Create hooks for:
    - `useLeaveRequests(filter)` -- useQuery, key: ['hr', 'leave', 'requests', filter]
    - `useLeaveRequest(id)` -- useQuery, key: ['hr', 'leave', 'requests', id]
    - `useLeaveBalance()` -- useQuery, key: ['hr', 'leave', 'balance']
    - `useEmployeeLeaveBalance(userId)` -- useQuery, enabled when userId provided
    - `useLeaveTypes()` -- useQuery, key: ['hr', 'leave', 'types'], staleTime: 5min
    - `useCreateLeaveRequest()` -- useMutation, invalidates leave requests + balance
    - `useApproveLeaveRequest()` -- useMutation, invalidates leave requests
    - `useRejectLeaveRequest()` -- useMutation, invalidates leave requests
    - `useCancelLeaveRequest()` -- useMutation, invalidates leave requests + balance
    - `useRecordSickLeave()` -- useMutation, invalidates leave requests

    - `useWorkTimeStatus()` -- useQuery, key: ['hr', 'time', 'status'], refetchInterval: 30000 (30s polling for real-time status in header)
    - `useActiveShift()` -- useQuery, key: ['hr', 'time', 'active']
    - `useClockIn()` -- useMutation, invalidates time status + active shift. On success, check compliance result and show toast:
      - If restViolation: toast.warning with rest hours message
    - `useClockOut()` -- useMutation, invalidates time status + active shift + entries. On success, check compliance result:
      - severity 'info': toast.info("8 Stunden erreicht")

      - severity 'error': toast.error("Arbeitszeitgrenze (10h) ueberschritten!")
    - `useStartBreak()` / `useEndBreak()` -- useMutation, invalidates active shift
    - `useWorkTimeEntries(filter)` -- useQuery
    - `useDailySummary(date)` -- useQuery
    - `useWeeklySummary(weekStart)` -- useQuery
    - `useSubmitCorrection()` -- useMutation
    - `useApproveCorrection()` -- useMutation

    - `useAbsenceCalendar(startDate, endDate, department?)` -- useQuery, key: ['hr', 'absences', 'calendar', ...]

    - `useEmployees(filter)` -- useQuery
    - `useSelfProfile()` -- useQuery, key: ['hr', 'employees', 'me']
    - `useEmployee(id)` -- useQuery
    - `useUpdateEmployee()` -- useMutation, invalidates employee list
    - `useUpdateSelfProfile()` -- useMutation, invalidates self profile
    - `useEmployeeDocuments(employeeId)` -- useQuery
    - `useUploadEmployeeDocument()` -- useMutation, invalidates documents

    - `useUpdateHRSettings()` -- useMutation
