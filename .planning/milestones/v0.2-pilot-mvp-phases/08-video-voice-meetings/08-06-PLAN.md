    ```typescript
    // Enums
    export type CallType = 'one_to_one' | 'group';
    export type CallStatus = 'ringing' | 'active' | 'ended';
    export type MeetingStatus = 'scheduled' | 'in_progress' | 'completed' | 'cancelled';
    export type RecordingStatus = 'active' | 'processing' | 'completed' | 'failed' | 'deleted';
    export type PresenceLevel = 'online' | 'away' | 'dnd' | 'in_call' | 'offline';
    export type RsvpStatus = 'pending' | 'accepted' | 'declined' | 'tentative';

    // Call types
    export interface CallSession { id, call_type, status, room_name, initiator_id, channel_id?, created_at, ended_at?, duration_seconds?, participants? }
    export interface CallParticipant { call_id, user_id, joined_at, left_at?, has_video, has_audio }
    export interface CreateCallRequest { call_type, target_user_ids, channel_id? }
    export interface JoinCallResponse { token, ws_url }

    // Meeting types
    export interface Meeting { id, title, description?, agenda?, organizer_id, status, scheduled_start, scheduled_end, actual_start?, actual_end?, room_name?, calendar_event_id?, recurring_meeting_id?, attendees?, created_at, updated_at }
    export interface MeetingAttendee { meeting_id, user_id, rsvp_status, user_name? }
    export interface CreateMeetingRequest { title, description?, agenda?, scheduled_start, scheduled_end, attendee_ids, calendar_event_id?, recurring_meeting_id? }
    export interface UpdateMeetingRequest { title?, description?, agenda?, scheduled_start?, scheduled_end? }
    export interface MeetingFilter { organizer_id?, status?, start_after?, start_before?, limit?, offset? }
    export interface MeetingSummary { meeting_id, duration_minutes, attendee_count, notes_summary, action_items }
    export interface StartMeetingResponse { meeting, token, ws_url }

    // Notes types
    export interface MeetingNotes { id, meeting_id, author_id, content, is_private, created_at, updated_at }
    export interface SaveNotesRequest { content, is_private }

    // Action items
    export interface ActionItem { id, meeting_id, description, assignee_id?, is_completed, task_id?, sort_order, created_at }
    export interface CreateActionItemRequest { description, assignee_id? }
    export interface ConvertActionItemsRequest { project_id }
    export interface ConvertActionItemsResponse { task_ids }

    // Recording types
    export interface Recording { id, call_id?, meeting_id?, status, egress_id?, file_url?, file_size_bytes?, duration_seconds?, retention_expires_at?, created_at }
    export interface RecordingConsent { recording_id, user_id, consented, responded_at }
    export interface RecordingConsentStatus { all_responded, consents }

    // Presence types
    export interface UserPresence { user_id, status: PresenceLevel, manual_override, last_activity? }
    export interface PresenceConfig { away_timeout_seconds }

    // Reaction types
    export interface Reaction { message_id, user_id, emoji, created_at }
    export interface ReactionSummary { message_id, emoji, count, user_ids, current_user_reacted }
    export interface ToggleReactionRequest { message_id, emoji }
    export interface ToggleReactionResponse { reactions: ReactionSummary[], added }
    ```

    **Create `useVideo.ts`:**
    Follow patterns from `useCalendars.ts` / `useEvents.ts`.
    ```typescript
    // Queries
    export const useActiveCalls = () => useQuery({ queryKey: ['calls', 'active'], queryFn: listActiveCalls });
    export const useCall = (callId: string) => useQuery({ queryKey: ['calls', callId], queryFn: () => getCall(callId), enabled: !!callId });
    export const useRecordings = (params: { callId?: string; meetingId?: string }) =>
      useQuery({ queryKey: ['recordings', params], queryFn: () => listRecordings(params), enabled: !!params.callId || !!params.meetingId });
    export const useRecordingConsent = (recordingId: string) =>
      useQuery({ queryKey: ['recordings', recordingId, 'consent'], queryFn: () => getRecordingConsent(recordingId), enabled: !!recordingId });

    // Mutations
    export const useCreateCall = () => useMutation({ mutationFn: createCall, onSuccess: () => queryClient.invalidateQueries({ queryKey: ['calls'] }) });
    export const useJoinCall = () => useMutation({ mutationFn: (callId: string) => joinCall(callId) });
    export const useEndCall = () => useMutation({ mutationFn: (callId: string) => endCall(callId), onSuccess: () => queryClient.invalidateQueries({ queryKey: ['calls'] }) });
    export const useStartRecording = () => useMutation({ mutationFn: (callId: string) => startRecording(callId) });
    export const useStopRecording = () => useMutation({ mutationFn: (callId: string) => stopRecording(callId) });
    export const useSetRecordingConsent = () => useMutation({ mutationFn: ({ recordingId, consented }: { recordingId: string; consented: boolean }) => setRecordingConsent(recordingId, consented) });
    ```
