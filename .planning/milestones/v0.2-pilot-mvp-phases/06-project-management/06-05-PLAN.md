**desktop/src/renderer/src/api/hooks/useProjects.ts:**
TanStack Query hooks following useDeals.ts pattern:

```typescript
// Queries
export function useProjects() -- GET /api/v1/projects, returns ProjectWithDetails[]
export function useProject(id: string) -- GET /api/v1/projects/{id}
export function useProjectMembers(projectId: string) -- GET /api/v1/projects/{id}/members
export function useProjectStatuses(projectId: string) -- GET /api/v1/projects/{id}/statuses
export function useProjectPreference(projectId: string) -- GET /api/v1/projects/{id}/preferences

// Mutations
export function useCreateProject() -- POST /api/v1/projects, invalidates ['projects']
export function useUpdateProject() -- PUT /api/v1/projects/{id}, invalidates ['projects', 'project', id]
export function useArchiveProject() -- DELETE /api/v1/projects/{id}, invalidates ['projects']
export function useAddMember() -- POST /api/v1/projects/{id}/members
export function useRemoveMember() -- DELETE /api/v1/projects/{id}/members/{userId}
export function useCreateStatus() -- POST /api/v1/projects/{id}/statuses
export function useUpdateStatus() -- PUT /api/v1/projects/{id}/statuses/{statusId}
export function useDeleteStatus() -- DELETE /api/v1/projects/{id}/statuses/{statusId}
export function useReorderStatuses() -- PUT /api/v1/projects/{id}/statuses/reorder
export function useSetPreference() -- PUT /api/v1/projects/{id}/preferences
```

Use the existing API client pattern (apiClient.get, apiClient.post, etc.) from the codebase.
